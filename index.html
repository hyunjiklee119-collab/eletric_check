Google Apps Script (Code.gs) 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "제출일시", "점검일자", "점검자", "점검부서",
        "관리번호", "기기종류", "제조년월", "세부사용위치",
        "1.전선/플러그", "2.콘센트/멀티탭", "3.가연물이격", "4.전두/과열방지", "5.전원차단/타이머", "6.청결/먼지", "7.소화용구",
        "기기별판정", "종합비고"
      ]);
    }
    
    var timestamp = data.timestamp || new Date().toLocaleString("ko-KR");
    var inspectionDate = data.inspectionDate || "";
    var inspectorName = data.inspectorName || "";
    var inspectionDept = data.inspectionDept || "";
    var overallRemarks = data.overallRemarks || "";
    var devices = data.devicesRawData || [];
    
    if (devices.length > 0) {
      devices.forEach(function(dev) {
        var chk = dev.checklist || {};
        var isOver5 = false;
        if (dev.mfgYM && inspectionDate) {
          var inspD = new Date(inspectionDate);
          var mfgParts = dev.mfgYM.split('-').map(Number);
          var mfgD = new Date(mfgParts[0], mfgParts[1] - 1, 1);
          var diffMonths = (inspD.getFullYear() - mfgD.getFullYear()) * 12 + (inspD.getMonth() - mfgD.getMonth());
          if (diffMonths > 60) isOver5 = true;
        }
        var isDefect = isOver5 || Object.values(chk).indexOf("불량") !== -1;
        var devStatus = isDefect ? (isOver5 ? "불량(5년초과)" : "불량(시정필요)") : "양호";
        
        // WHE- 접두사와 입력된 숫자를 합쳐서 시트에 기록
        var fullMgmtNum = "WHE-" + (dev.mgmtNum || "");
        
        var rowData = [
          timestamp, inspectionDate, inspectorName, inspectionDept,
          fullMgmtNum, dev.deviceType || "", dev.mfgYM || "", dev.subLocation || "",
          chk.item1 || "양호", chk.item2 || "양호", chk.item3 || "양호", chk.item4 || "양호",
          chk.item5 || "양호", chk.item6 || "양호", chk.item7 || "양호",
          devStatus, overallRemarks
        ];
        sheet.appendRow(rowData);
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
